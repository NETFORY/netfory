const job = window.smartholdem.useAI({
  prompt: 'Твой вопрос…',
  model: 'tinyllama',      // опционально: alias модели
  maxTokens: 256,           // опционально
  onToken: (t) => out.textContent += t,  // стриминг
})
const { text, node, ms } = await job.done  // финал
job.cancel()                    // отмена в любой момент
// доступность: window.__SMARTNET__.features.ai === true
